import CryptoLean.Group.ShiftRepresentations
import Mathlib.Probability.Moments.Variance
import Mathlib.Probability.UniformOn
import Mathlib.MeasureTheory.Integral.Pi
import Mathlib.MeasureTheory.Integral.Bochner.SumMeasure

/-! Exact moments of the actual compatible two-gate count, under the raw
independent uniform auxiliary-pool law, for a fixed symmetric filter. -/
set_option autoImplicit false

noncomputable section

open MeasureTheory ProbabilityTheory
open scoped BigOperators

namespace CryptoLean.ShiftRepresentations

variable {G : Type*} [AddCommGroup G] [Fintype G] [DecidableEq G]
  [MeasurableSpace G] [DiscreteMeasurableSpace G] {k : ℕ}

theorem shifted_hit_mean (U : G) (S : Finset G) :
    (∫ g, (if U + g ∈ S then (1 : ℝ) else 0) ∂uniformOn (Set.univ : Set G)) =
      (S.card : ℝ) / Fintype.card G := by
  have hweight (g : G) :
      (uniformOn (Set.univ : Set G)).real {g} = 1 / (Fintype.card G : ℝ) := by
    simp [measureReal_def, uniformOn_univ]
  have hsum : (∑ g : G, if U + g ∈ S then (1 : ℝ) else 0) = S.card := by
    calc
      _ = ∑ g : G, if g ∈ S then (1 : ℝ) else 0 :=
        Fintype.sum_equiv (Equiv.addLeft U) _ _ (fun _ ↦ rfl)
      _ = S.card := by simp
  rw [integral_fintype ((MemLp.of_discrete (p := 2)).integrable one_le_two)]
  simp only [hweight, smul_eq_mul, ← Finset.mul_sum, hsum]
  ring

theorem shifted_hit_variance (U : G) (S : Finset G) :
    Var[fun g ↦ if U + g ∈ S then (1 : ℝ) else 0; uniformOn (Set.univ : Set G)] =
      ((S.card : ℝ) / Fintype.card G) * (1 - (S.card : ℝ) / Fintype.card G) := by
  have hsquare : ((fun g : G ↦ if U + g ∈ S then (1 : ℝ) else 0) ^ 2) =
      (fun g ↦ if U + g ∈ S then (1 : ℝ) else 0) := by
    funext g
    simp only [Pi.pow_apply]
    split_ifs <;> norm_num
  rw [variance_eq_sub MemLp.of_discrete, hsquare, shifted_hit_mean]
  ring

theorem acceptedCount_real_eq_sum (U V : G) (S : Finset G) (T : Fin k → G)
    (hzero : U + V = 0) (hsym : ∀ g : G, g ∈ S ↔ -g ∈ S) :
    (acceptedCount U V S T : ℝ) =
      ∑ j : Fin k, if U + T j ∈ S then (1 : ℝ) else 0 := by
  rw [acceptedCount_eq_sum U V S T hzero hsym]
  simp only [Nat.cast_sum, Nat.cast_ite, Nat.cast_one, Nat.cast_zero]

theorem acceptedCount_function_eq_sum (U V : G) (S : Finset G)
    (hzero : U + V = 0) (hsym : ∀ g : G, g ∈ S ↔ -g ∈ S) :
    (fun T : Fin k → G ↦ (acceptedCount U V S T : ℝ)) =
      ∑ j : Fin k, fun T : Fin k → G ↦ if U + T j ∈ S then (1 : ℝ) else 0 := by
  funext T
  simpa only [Finset.sum_apply] using acceptedCount_real_eq_sum U V S T hzero hsym

theorem acceptedCount_mean (U V : G) (S : Finset G)
    (hzero : U + V = 0) (hsym : ∀ g : G, g ∈ S ↔ -g ∈ S) :
    (∫ T : Fin k → G, (acceptedCount U V S T : ℝ)
      ∂Measure.pi (fun _ : Fin k ↦ uniformOn (Set.univ : Set G))) =
        (k : ℝ) * ((S.card : ℝ) / Fintype.card G) := by
  simp_rw [acceptedCount_real_eq_sum U V S _ hzero hsym]
  rw [integral_finset_sum _ (fun _ _ ↦ (MemLp.of_discrete (p := 2)).integrable one_le_two)]
  simp only [integral_comp_eval (MemLp.of_discrete (p := 2)).aestronglyMeasurable,
    shifted_hit_mean, Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_eq_mul]

theorem acceptedCount_variance (U V : G) (S : Finset G)
    (hzero : U + V = 0) (hsym : ∀ g : G, g ∈ S ↔ -g ∈ S) :
    Var[fun T : Fin k → G ↦ (acceptedCount U V S T : ℝ);
      Measure.pi (fun _ : Fin k ↦ uniformOn (Set.univ : Set G))] =
        (k : ℝ) * ((S.card : ℝ) / Fintype.card G) *
          (1 - (S.card : ℝ) / Fintype.card G) := by
  rw [acceptedCount_function_eq_sum U V S hzero hsym]
  rw [variance_sum_pi (fun _ ↦ MemLp.of_discrete)]
  simp only [shifted_hit_variance, Finset.sum_const, Finset.card_univ,
    Fintype.card_fin, nsmul_eq_mul, mul_assoc]

theorem acceptedCount_second_moment (U V : G) (S : Finset G)
    (hzero : U + V = 0) (hsym : ∀ g : G, g ∈ S ↔ -g ∈ S) :
    (∫ T : Fin k → G, (acceptedCount U V S T : ℝ) ^ 2
      ∂Measure.pi (fun _ : Fin k ↦ uniformOn (Set.univ : Set G))) =
        (k : ℝ) * ((S.card : ℝ) / Fintype.card G) +
          (k : ℝ) * ((k : ℝ) - 1) * ((S.card : ℝ) / Fintype.card G) ^ 2 := by
  have hvar := variance_eq_sub
    (MemLp.of_discrete (f := fun T : Fin k → G ↦ (acceptedCount U V S T : ℝ))
      (p := 2) (μ := Measure.pi (fun _ : Fin k ↦ uniformOn (Set.univ : Set G))))
  rw [acceptedCount_variance U V S hzero hsym, acceptedCount_mean U V S hzero hsym] at hvar
  change (k : ℝ) * ((S.card : ℝ) / Fintype.card G) *
      (1 - (S.card : ℝ) / Fintype.card G) =
    (∫ T : Fin k → G, (acceptedCount U V S T : ℝ) ^ 2
      ∂Measure.pi (fun _ : Fin k ↦ uniformOn (Set.univ : Set G))) -
        ((k : ℝ) * ((S.card : ℝ) / Fintype.card G)) ^ 2 at hvar
  nlinarith [hvar]

end CryptoLean.ShiftRepresentations
